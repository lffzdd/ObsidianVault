```python
# ============================================================
# Thompson NFA —— 第二步：引入 Frag + 三个组合子，自动建图
# ============================================================
#
# 整体分两个阶段：
#   1. 建图阶段：把 pattern 变成一张由节点(State)组成的有向图
#   2. 跑图阶段：拿 text 在图上逐字符推进，看能不能走到终点
#
# 上一版是手工接线，只能造 "a" 这一个 pattern。
# 这一版引入 Frag（半成品电路）和三个组合子（concat/alternate/star），
# 可以用代码自动拼出任意 pattern 的图。
# ============================================================

from typing import List, Literal, Optional, Tuple, Union

# ============================================================
# 第一阶段：建图（只跟 pattern 有关，和 text 无关）
# ============================================================

# ---- 节点 ----

SPLIT = -1   # 分裂态：不吃字符，ε 分叉走两条路
MATCH = -2   # 接受态：终点

Symbol = Union[str, int]                         # 字符，或 SPLIT/MATCH 这种哨兵值
PatchSite = Tuple["State", Literal["out", "out1"]]  # 一个还没接上的 State 字段

class State:
    """
    图里的一个节点。

    三种形态：
    - 字符态：c 是一个字符（如 'a'），表示"要消耗这个字符才能通过"
    - 分裂态：c == SPLIT，不消耗字符，同时走 out 和 out1 两条路
    - 接受态：c == MATCH，终点，没有出边
    """
    def __init__(
        self,
        c: Symbol,
        out: Optional["State"] = None,
        out1: Optional["State"] = None,
    ):
        self.c: Symbol = c                  # 通行条件：什么字符能通过（或 SPLIT/MATCH）
        self.out: Optional["State"] = out   # 出边1：通过后走到哪个节点
        self.out1: Optional["State"] = out1 # 出边2：只有分裂态才用

# ---- 半成品电路 (Frag) ----

class Frag:
    """
    建图过程中的半成品——一小段图，有入口，但出口还没接上。

    打个比方：一截电线，左边（start）已经焊好了，
    右边（outs）还悬着，等着后面来的东西焊上去。

    属性：
        start: 这段图的入口节点，类型是 State
        outs:  悬空线头列表，每个元素是 (node, 字段名)
               类型是 List[Tuple[State, Literal["out", "out1"]]]
               意思是"node 的这个字段（'out' 或 'out1'）还是 None，等着被焊上"
    """
    def __init__(self, start: State, outs: List[PatchSite]):
        self.start: State = start
        self.outs: List[PatchSite] = outs

def patch(outs: List[PatchSite], target: State) -> None:
    """
    焊接：把一堆悬空线头全部接到 target 节点上。

    例如 outs = [(node_a, 'out'), (node_b, 'out1')]
    执行后：node_a.out = target, node_b.out1 = target
    """
    for node, attr in outs:
        setattr(node, attr, target)

# ---- 基本构件：单个字符 ----

def literal(c):
    """
    为 pattern 里的单个字符 c 建一个 Frag。

        ┌─────┐
        │  c  │ ═══→ ???（悬空）
        └─────┘

    入口是这个节点本身，唯一的悬空线头是它的 out。
    """
    node = State(c)                       # 造节点，out 先空着
    return Frag(node, [(node, 'out')])     # 打包成半成品

# ---- 组合子 1：串联 (concat) ----

def concat(f1: Frag, f2: Frag) -> Frag:
    """
    把两个 Frag 串联起来：f1 后面接 f2。

    做法：把 f1 所有悬空线头焊到 f2 的入口。

    之前：
        f1: [a] ═══→ ???       f2: [b] ═══→ ???

    之后：
        [a] ──→ [b] ═══→ ???

    新 Frag 的入口是 f1 的入口，悬空线头是 f2 的悬空线头。
    """
    patch(f1.outs, f2.start)              # f1 的线头全接到 f2 的入口
    return Frag(f1.start, f2.outs)        # 入口不变，线头换成 f2 的

# ---- 组合子 2：选择 (alternate) ----

def alternate(f1: Frag, f2: Frag) -> Frag:
    """
    f1 | f2：造一个分裂态，一条路去 f1，一条路去 f2。

    之前：
        f1: [a] ═══→ ???       f2: [b] ═══→ ???

    之后：
                ┌→ [a] ═══→ ???
        [SPLIT]─┤
                └→ [b] ═══→ ???

    新 Frag 的入口是分裂态，悬空线头是 f1 + f2 的线头合并。
    """
    split = State(SPLIT, f1.start, f2.start)   # 造分裂态，两条路
    return Frag(split, f1.outs + f2.outs)       # 两边的线头都还悬着

# ---- 组合子 3：重复 (star) ----

def star(f: Frag) -> Frag:
    """
    f*：匹配 0 次或多次。造一个分裂态：
    - 一条路进入 f（循环体），f 做完后回到分裂态（自环）
    - 一条路跳过 f（零次匹配）

    之前：
        f: [a] ═══→ ???

    之后：
              ┌→ [a] ──→─┐
        [SPLIT]─┤          ↑  （自环：a 做完后回到 SPLIT 再来一次）
              └→ ═══→ ???    （跳过：零次匹配，悬空等后面的东西）

    新 Frag 的入口是分裂态，悬空线头是分裂态的 out1（跳过那条路）。
    """
    split = State(SPLIT)                   # 造分裂态，两条路先空着
    split.out = f.start                    # 路1：进入循环体 f
    patch(f.outs, split)                   # f 做完后回到 split（自环）
    split.out1 = None                      # 路2：跳过（悬空，等后面的东西来焊）
    return Frag(split, [(split, 'out1')])   # 线头是"跳过"那条路

# ---- 收尾：把最后的悬空线头焊到终点 ----

def build(frag: Frag) -> State:
    """
    把一个 Frag 变成完整的 NFA：把所有剩余悬空线头接到 accept 节点。
    返回图的起点，可以直接丢给 simulate。
    """
    accept = State(MATCH)
    patch(frag.outs, accept)
    return frag.start

# ============================================================
# 第二阶段：跑图（只跟 text 有关，图已经建好了）
# ============================================================

def add_epsilon(node: Optional[State], state_set: set) -> None:
    """
    处理 ε 转移：如果 node 是分裂态(SPLIT)，不消耗字符，
    递归地把它的两条出边指向的节点都加入集合。
    """
    if node is None:
        return
    if node.c == SPLIT:
        add_epsilon(node.out,  state_set)
        add_epsilon(node.out1, state_set)
    else:
        state_set.add(node)

def simulate(start: State, text: str) -> bool:
    """
    在 NFA 图上跑一个 text，判断是否匹配。

    和上一版的区别：现在图里有分裂态(SPLIT)了，
    分裂态不吃字符，需要递归展开（ε 转移）。
    """
    # 起手：站在起点，但如果起点是 SPLIT 要展开
    current = set()
    add_epsilon(start, current)

    # 逐字符推进
    for ch in text:
        next_states = set()
        for node in current:
            if node.c == ch:
                add_epsilon(node.out, next_states)   # 走过去，遇到 SPLIT 继续展开
        current = next_states

    # text 全部读完，检查有没有人站在终点
    for node in current:
        if node.c == MATCH:
            return True
    return False

# ============================================================
# 测试：用组合子自动建图，然后跑
# ============================================================

print("=== pattern: 'a' ===")
nfa = build(literal('a'))
print('  "a"  =>', simulate(nfa, "a"))     # True
print('  "b"  =>', simulate(nfa, "b"))     # False
print('  ""   =>', simulate(nfa, ""))      # False

print()
print("=== pattern: 'ab'  (concat) ===")
nfa = build(concat(literal('a'), literal('b')))
print('  "ab" =>', simulate(nfa, "ab"))    # True
print('  "a"  =>', simulate(nfa, "a"))     # False
print('  "b"  =>', simulate(nfa, "b"))     # False
print('  "abc"=>', simulate(nfa, "abc"))   # False

print()
print("=== pattern: 'a|b'  (alternate) ===")
nfa = build(alternate(literal('a'), literal('b')))
print('  "a"  =>', simulate(nfa, "a"))     # True
print('  "b"  =>', simulate(nfa, "b"))     # True
print('  "c"  =>', simulate(nfa, "c"))     # False
print('  "ab" =>', simulate(nfa, "ab"))    # False

print()
print("=== pattern: 'a*'  (star) ===")
nfa = build(star(literal('a')))
print('  ""   =>', simulate(nfa, ""))      # True  (零次)
print('  "a"  =>', simulate(nfa, "a"))     # True  (一次)
print('  "aaa"=>', simulate(nfa, "aaa"))   # True  (多次)
print('  "b"  =>', simulate(nfa, "b"))     # False

print()
print("=== pattern: 'a*b'  (star + concat) ===")
nfa = build(concat(star(literal('a')), literal('b')))
print('  "b"  =>', simulate(nfa, "b"))     # True  (a 零次)
print('  "ab" =>', simulate(nfa, "ab"))    # True  (a 一次)
print('  "aab"=>', simulate(nfa, "aab"))   # True  (a 两次)
print('  "aa" =>', simulate(nfa, "aa"))    # False (没有 b)

print()
print("=== pattern: 'a|bc'  (alternate + concat) ===")
nfa = build(alternate(literal('a'), concat(literal('b'), literal('c'))))
print('  "a"  =>', simulate(nfa, "a"))     # True
print('  "bc" =>', simulate(nfa, "bc"))    # True
print('  "b"  =>', simulate(nfa, "b"))     # False
```
