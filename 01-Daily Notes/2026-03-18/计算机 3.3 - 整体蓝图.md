### 🏛️ 蓝图一：浏览器主进程 (Browser Process)

**角色定位：** 整个浏览器的大脑与总调度室。它不运行任何前端 JS 代码，只负责与操作系统打交道、管理所有标签页、控制网络硬件和定时器硬件。

```C++
// =====================================================================
// BrowserProcess.cpp (浏览器主进程)
// =====================================================================

class BrowserProcess {
    IPC_Channel ipc_channel;      // 负责与渲染进程通信的管道
    NetworkService network;       // 负责真实 HTTP 请求的底层硬件调用
    TimerSystem timer_system;     // 负责调用操作系统底层的时钟
    UI_Thread ui_thread; // 专门负责绘制浏览器外壳（地址栏、标签栏）的线程

public:
    // 【核心入口】浏览器主进程的永动机
    void mainLoop() {
        while (true) {
            // 阻塞等待：等待来自操作系统、硬件或渲染进程的任何消息
            Message msg = waitForAnyEvent(); 

            switch (msg.type) {
                // 1. 用户在地址栏敲击回车
                case OS_EVENT_UI_INPUT:
                    string url = msg.text;
                    RendererProcess* renderer = createOrGetRenderer(url);
                    network.startRequest(url, renderer);
                    break;

                // 2. 底层网络下载到了数据分片 (Chunk)
                case NETWORK_DATA_ARRIVED:
                    // 主进程不解析 HTML，直接把数据通过 IPC 灌给渲染进程
                    ipc_channel.send(msg.targetRenderer, {
                        type: "IPC_HTML_CHUNK",
                        data: msg.bytes
                    });
                    break;

                // 3. 处理渲染进程通过 IPC 发来的各种“求助”请求
                case IPC_FROM_RENDERER:
                    handleRendererRequest(msg);
                    break;

                // 4. 操作系统底层的定时器到期了
                case OS_TIMER_FIRED:
                    // 通知对应的渲染进程：你的闹钟响了
                    ipc_channel.send(msg.targetRenderer, {
                        type: "IPC_TIMER_READY",
                        callbackId: msg.timerId
                    });
                    break;
            }
        }
    }

private:
    // 分发渲染进程的求助请求
    void handleRendererRequest(Message msg) {
        if (msg.intent == "PLEASE_FETCH_DATA") {
            // 替渲染进程发起网络请求
            network.startRequest(msg.url, msg.sourceRenderer, [this](Response res) {
                // 请求完成后，把结果通过 IPC 丢回给渲染进程
                ipc_channel.send(msg.sourceRenderer, {
                    type: "IPC_FETCH_DONE",
                    data: res.body,
                    promiseId: msg.promiseId
                });
            });
        } 
        else if (msg.intent == "PLEASE_SET_TIMEOUT") {
            // 替渲染进程在操作系统层面注册一个定时器
            timer_system.register(msg.delayMs, msg.sourceRenderer, msg.timerId);
        }
    }
};
```

// =====================================================================
// BrowserProcess.cpp (浏览器主进程 2.0 - 完整生命周期)
// =====================================================================

class BrowserProcess {
    IPC_Server ipc_server;        // 负责监听渲染进程的连接请求
    NetworkService network;       // 独立于主线程的网络服务
    TimerSystem timer_system;     // 硬件时钟接口
    UI_Thread ui_thread;          // 专门负责绘制浏览器外壳（地址栏、标签栏）的线程

public:
    // 【系统入口】当你在桌面双击 Chrome 图标时，执行这里
    void start() {
        // =========================================
        // 阶段一：点火与初始化 (Initialization)
        // =========================================
        
        // 1. 初始化底层的 IPC 服务，打开 Socket/管道，开始监听
        ipc_server.listen(); 
        
        // 2. 启动独立运行的后台线程（网络、存储等）
        network.startBackgroundThread();
        
        // 3. 极其关键：初始化并启动 UI 线程
        // 这会向操作系统注册窗口，并挂载系统的硬件输入钩子
        ui_thread.initializeAndShowBrowserWindow();

        // 4. 注册操作系统级别的事件监听器（注册完毕后，OS 才会把事件发给本进程）
        registerSystemHooks();

        // =========================================
        // 阶段二：进入主循环 (Message Pump)
        // =========================================
        mainLoop();
    }

private:
    // 【核心调度室】现在，系统事件终于源源不断地进来了
    void mainLoop() {
        while (true) {
            // 此时的 wait 就不再是死等了。
            // 它是底层 OS 提供的阻塞调用 (如 Linux 的 epoll_wait 或 Windows 的 GetMessage)
            // 当 OS 队列中有鼠标点击，或者 IPC 管道有数据到来时，OS 会唤醒这个线程。
            Message msg = waitForAnyEvent(); 

            switch (msg.type) {
                // UI 线程捕获到了地址栏的键盘敲击，打包成消息发给了主循环
                case OS_EVENT_UI_INPUT:
                    string url = msg.text;
                    RendererProcess* renderer = createOrGetRenderer(url);
                    network.startRequest(url, renderer);
                    break;

                case NETWORK_DATA_ARRIVED:
                    ipc_server.send(msg.targetRenderer, {
                        type: "IPC_HTML_CHUNK",
                        data: msg.bytes
                    });
                    break;

                case IPC_FROM_RENDERER:
                    handleRendererRequest(msg);
                    break;

                case OS_TIMER_FIRED:
                    ipc_server.send(msg.targetRenderer, {
                        type: "IPC_TIMER_READY",
                        callbackId: msg.timerId
                    });
                    break;
            }
        }
    }

    void handleRendererRequest(Message msg) {
        // ... (与上一版相同，处理 fetch 和 setTimeout 等)
    }
};

### 🏭 蓝图二：渲染进程 (Renderer Process)

**角色定位：** 每个标签页专属的“车间”。包含 V8 引擎、渲染引擎（Blink）以及大名鼎鼎的“事件循环”。它没有权限直接访问网络或系统硬件，只能发 IPC 消息求助。

```C++
// =====================================================================
// RendererProcess.cpp (渲染进程 - 你写的代码都在这里跑)
// =====================================================================

class RendererProcess {
    IPC_Channel ipc_channel;      // 负责与主进程通信的管道
    V8Engine js_engine;           // JS 解释与执行器
    HtmlParser parser;            // HTML 解析器
    
    // 【核心数据结构】事件循环的任务队列
    Queue<Task> macrotaskQueue;   // 宏任务队列 (setTimeout, script等)
    Queue<Task> microtaskQueue;   // 微任务队列 (Promise.then等)

public:
    // 【核心入口】渲染进程主线程的永动机（这就是常说的 Event Loop）
    void mainThreadLoop() {
        while (true) {
            // 1. 接收来自主进程的 IPC 消息，并将其转化为本地任务
            processIPCAndGenerateTasks();

            // 2. 执行宏任务 (每次只拿一个)
            if (!macrotaskQueue.empty()) {
                Task currentMacroTask = macrotaskQueue.pop();
                executeTask(currentMacroTask); 
            }

            // 3. 清空微任务 (只要队列不空，就一直执行)
            while (!microtaskQueue.empty()) {
                Task currentMicroTask = microtaskQueue.pop();
                js_engine.run(currentMicroTask.code);
            }

            // 4. 渲染流水线 (根据屏幕刷新率 60Hz，判断是否需要绘制)
            if (needsRender()) {
                updateStyle();
                layout();
                paint();
            }
        }
    }

private:
    // 专门处理外部发来的 IPC 消息，并向队列中“塞”任务
    void processIPCAndGenerateTasks() {
        Message msg = ipc_channel.receive();
        if (!msg) return;

        switch (msg.type) {
            // 收到主进程传来的 HTML 数据
            case "IPC_HTML_CHUNK":
                // 安排一个解析 HTML 的宏任务
                macrotaskQueue.push( Task(PARSE_HTML, msg.data) );
                break;

            // 主进程通知：定时器时间到了
            case "IPC_TIMER_READY":
                // 安排一个执行定时器回调的宏任务
                macrotaskQueue.push( Task(RUN_JS_CALLBACK, msg.callbackId) );
                break;

            // 主进程通知：网络请求数据回来了
            case "IPC_FETCH_DONE":
                // 安排一个 Promise resolve 的微任务！(注意这里是微任务)
                microtaskQueue.push( Task(RESOLVE_PROMISE, msg.promiseId, msg.data) );
                break;
        }
    }

    // 真正执行任务的逻辑
    void executeTask(Task t) {
        if (t.type == PARSE_HTML) {
            // 解析 HTML 时如果遇到 <script>，会立刻挂起解析，并调用 V8
            parser.parse(t.data); 
        } 
        else if (t.type == RUN_JS_CALLBACK) {
            // 将 JS 回调函数压入 V8 的调用栈执行
            js_engine.run(t.callbackCode);
        }
    }

    // --- 【宿主 API 暴露给 JS 的底层实现】 ---
    
    // 当你的 JS 代码调用 setTimeout 时，实际执行的是这个 C++ 函数
    public void hostApi_setTimeout(Function cb, int delayMs) {
        int timerId = generateId();
        saveCallbackInMemory(timerId, cb); // 把函数暂存在渲染进程内存中
        
        // 渲染进程自己不倒计时，而是通过 IPC 委托主进程
        ipc_channel.sendToBrowser({
            intent: "PLEASE_SET_TIMEOUT",
            delayMs: delayMs,
            timerId: timerId
        });
        // 函数立刻 return，主线程继续往下跑
    }

    // 当你的 JS 代码调用 fetch 时，实际执行的是这个 C++ 函数
    public void hostApi_fetch(String url, Promise p) {
        int promiseId = generateId();
        savePromiseInMemory(promiseId, p); // 暂存 Promise 状态
        
        // 通过 IPC 委托主进程去发网络请求
        ipc_channel.sendToBrowser({
            intent: "PLEASE_FETCH_DATA",
            url: url,
            promiseId: promiseId
        });
        // 函数立刻 return，主线程继续往下跑
    }
};
```

### 🧩 蓝图合体：一行代码的时空旅行

当你在这个架构下执行 `setTimeout(() => console.log('A'), 1000)` 时，它跨越两个进程的完整生命周期如下：

1. **角色定位：** 主线程正在执行一段 JS 宏任务，遇到了 `setTimeout`。
2. **角色定位：** 主线程调用底层 C++ `hostApi_setTimeout`，向 `IPC_Channel` 塞入一条消息：`PLEASE_SET_TIMEOUT`。**[Renderer]**。
3. **[Renderer]** 主进程的 `mainLoop` 截获了这条 IPC 消息，调用操作系统底层的硬件时钟，设定一个 1000ms 的倒计时。
4. _(1000ms 过去了...)_
5. **渲染进程主线程不等待，继续往下执行 JS** 操作系统硬件中断触发，主进程收到 `OS_TIMER_FIRED`，立刻通过 IPC 向渲染进程发送：`IPC_TIMER_READY`。
6. **[Browser]** 渲染进程的主线程在下一次 `while(true)` 循环中，解析到了这条 IPC 消息，于是把包装好的回调函数推入 **[Browser]**。
7. **[Renderer]** 等到主线程当前的调用栈空了，且微任务队列也清空了，事件循环从 `macrotaskQueue` 捞出这个任务，交给 V8 引擎执行，控制台终于打印出了 `macrotaskQueue`。

通过这两段伪代码，你能清晰地看到：**`'A'`**、**执行者（渲染进程 JS 引擎）** [Renderer] **通信桥梁（IPC）** 的楚河汉界。
