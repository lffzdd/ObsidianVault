$$(k+1)^{p+1}-k^{p+1}=C_{p+1}^1 k^p + C_{p+1}^2 k^{p-1} + \dots + C_{p+1}^{p+1} k^0$$
对左边求和：
$$\sum_{k=1}^n\big((k+1)^{p+1}-k^{p+1}\big)=(n+1)^{p+1}-1$$
对右边求和：
$$
\begin{align}
&\sum_{k=1}^n\big(C_{p+1}^1 k^p + C_{p+1}^2 k^{p-1} + \dots + C_{p+1}^{p+1} k^0\big) \\
=&C_{p+1}^1 S_p(n) + C_{p+1}^2 S_{p-1}(n) + \dots + C_{p+1}^{p+1} S_0(n)
\end{align}
$$
因此：
$$
\begin{align}
S_p(n)&=\frac{1}{C_{p+1}^1}\big((n+1)^{p+1}-1-C_{p+1}^2 S_{p-1}(n) - \dots - C_{p+1}^{p+1} S_0(n)\big) \\
&= \frac{1}{p+1}\big((n+1)^{p+1}-1-C_{p+1}^2 S_{p-1}(n) - \dots - C_{p+1}^{p+1} S_0(n)\big)
\end{align}
$$

回过头：
$$
\begin{align}
(p+1)S_p(n) &= (n+1)^{p+1}-1-C_{p+1}^2 S_{p-1}(n) - \dots - C_{p+1}^{p+1} S_{0(n)}\\
(p)S_{p-1}(n) &= (n+1)^{p}-1-C_{p}^2 S_{p-2}(n) - \dots - C_{p}^{p} S_{0(n)}\\
\cdots \\
(2)S_{1}(n) &= (n+1)^{2}-1-C_{2}^2 S_{0}(n) \\
(1)S_{0}(n) &= (n+1)^{1}-1
\end{align}
$$
$S_{0}(n)$ 是 $1$ 次多项式，$S_{1}(n)$ 是 $2$ 次多项式，$\dots$，$S_{p-1}(n)$ 是 $p$ 次多项式，因此 $(p+1)S_p(n)$ 是 $p+1$ 次多项式。

所以
$$
S_p(n) = \frac{1}{p+1}n^{p+1} + c_1 n^p + c_2 n^{p-1} + \cdots
$$

现在来计算系数的值，
 1.  $n^{p+1}$,系数为 $\frac{1}{p+1}$

2. 对$n^p$,来源于$-C_{p+1}^2 S_{p-1}(n)$,
$$
S_{p-1}(n) = \frac{1}{p}(n+1)^p + 更低阶项
$$
因此 $-C_{p+1}^2 S_{p-1}(n)$ 中 $n^p$ 的系数为 $-\frac{C_{p+1}^2}{p}$

3. 对 $n^{p-1}$,来源于 $-C_{p+1}^2 S_{p-1}(n)$ 和 $-C_{p+1}^3 S_{p-2}(n)$,
$$
\begin{align}
S_{p-1}(n) &= \frac{1}{p}(n+1)^p - \frac{C_{p}^2}{p(p-1)}(n+1)^{p-1} + 更低阶项\\
S_{p-2}(n) &= \frac{1}{p-1}(n+1)^{p-1} + 更低阶项

\end{align}
$$
因此 $-C_{p+1}^2 S_{p-1}(n)$ 中 $n^{p-1}$ 的系数为 $\frac{C_{p+1}^2 C_{p}^2}{p(p-1)}$，$-C_{p+1}^3 S_{p-2}(n)$ 中 $n^{p-1}$ 的系数为 $-\frac{C_{p+1}^3}{p-1}$，所以 $n^{p-1}$ 的系数为 $\frac{C_{p+1}^2 C_{p}^2}{p(p-1)} - \frac{C_{p+1}^3}{p-1}$

4. 对 $n^{p-2}$,来源于 $-C_{p+1}^2 S_{p-1}(n)$, $-C_{p+1}^3 S_{p-2}(n)$ 和 $-C_{p+1}^4 S_{p-3}(n)$,
$$
\begin{align}
S_{p-1}(n) &= \frac{1}{p}(n+1)^p - \frac{C_{p}^2}{p(p-1)}(n+1)^{p-1} + \frac{C_{p}^3}{p(p-1)(p-2)}(n+1)^{p-2} + 更低阶项\\
S_{p-2}(n) &= \frac{1}{p-1}(n+1)^{p-1} + \frac{1}{2(p-1)}(n+1)^{p-2} + 更低阶项\\
S_{p-3}(n) &= \frac{1}{p-2}(n+1)^{p-2} + 更低阶项
\end{align}
$$
因此 $-C_{p+1}^2 S_{p-1}(n)$ 中 $n^{p-2}$ 的系数为 $-\frac{C_{p+1}^2 C_{p}^3}{p(p-1)(p-2)}$，$-C_{p+1}^3 S_{p-2}(n)$ 中 $n^{p-2}$ 的系数为 $-\frac{C_{p+1}^3}{2(p-1)}$，$-C_{p+1}^4 S_{p-3}(n)$ 中 $n^{p-2}$ 的系数为 $-\frac{C_{p+1}^4}{p-2}$，所以 $n^{p-2}$ 的系数为 $-\frac{C_{p+1}^2 C_{p}^3}{p(p-1)(p-2)} - \frac{C_{p+1}^3}{2(p-1)} - \frac{C_{p+1}^4}{p-2}$
